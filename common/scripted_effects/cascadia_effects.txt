CAS_nazi_anger_calculation_effect = {
	set_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = 0
	}
	if = {
		limit = {
			check_variable = { var_nazi_influence > 24 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_influence < 55 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 1
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_influence < 80 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 2
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_influence > 79 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 3
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_influence < 26 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_influence < 6 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level_anti
				value = 3
			}
		}
		else = {
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level_anti
				value = 1
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_satisfaction < 55 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 30 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 1
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 10 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 2
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_satisfaction < 11 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 3
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_satisfaction > 54 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 79 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level_anti
				value = 2
			}
		}
		else = {
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level_anti
				value = 1
			}
		}
	}
	add_to_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_satisfaction_rebellion_level
	}
	add_to_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_influence_rebellion_level
	}
	subtract_from_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_influence_rebellion_level_anti
	}
	subtract_from_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_satisfaction_rebellion_level_anti
	}
	clamp_variable = {
		var = CAS.var_nazi_rebellion_indicator
		min = 0
	}
}
