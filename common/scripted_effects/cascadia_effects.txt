CAS_nazi_coup_trigger_mission_effect = { 
	if = {
		limit = { NOT = { has_war_with = PRA } }
		if = {
			limit = {
				AND = {
					NOT = { 
						AND = { 
							has_country_flag = CAS_coup_progressing_flag 
							has_country_flag = CAS_nazis_purged_flag 
						} 
					}
					check_variable = { var_nazi_rebellion_indicator > 1 }
				} 
			}
			country_event = {
				id = cascadia_rework.5
				days = 3
			}
			set_var = {
				var = CAS.var_nazi_days_mission_coup_reduction
				value = CAS.var_nazi_rebellion_indicator
			}
			multiply_variable = {
				var = CAS.var_nazi_days_mission_coup_reduction
				value = -20
			}
		}
	}
}


CAS_nazi_anger_calculation_effect = {
	set_variable = {
		var = CAS.var_nazi_rebellion_indicator_prev
		value = CAS.var_nazi_rebellion_indicator
	}
	set_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = 0
	}
	if = {
		limit = {
			check_variable = { var_nazi_influence > 24 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_influence < 55 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 1
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_influence < 80 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 2
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_influence > 79 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level
				value = 3
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_influence < 26 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_influence < 6 }
			}
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level_anti
				value = 3
			}
		}
		else = {
			set_variable = {
				var = CAS.var_nazi_influence_rebellion_level_anti
				value = 1
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_satisfaction < 55 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 30 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 1
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 10 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 2
			}
		}
		else_if = {
			limit = {
				check_variable = { var_nazi_satisfaction < 11 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level
				value = 3
			}
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_satisfaction > 54 }
		}
		if = {
			limit = {
				check_variable = { var_nazi_satisfaction > 79 }
			}
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level_anti
				value = 2
			}
		}
		else = {
			set_variable = {
				var = CAS.var_nazi_satisfaction_rebellion_level_anti
				value = 1
			}
		}
	}
	add_to_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_satisfaction_rebellion_level
	}
	add_to_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_influence_rebellion_level
	}
	add_to_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_extra_negative_risk_coup
	}
	subtract_from_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_influence_rebellion_level_anti
	}
	subtract_from_variable = {
		var = CAS.var_nazi_rebellion_indicator
		value = CAS.var_nazi_satisfaction_rebellion_level_anti
	}
	clamp_variable = {
		var = CAS.var_nazi_rebellion_indicator
		min = 0
	}
	
	if = {
		limit = {
			check_variable = { var_nazi_rebellion_indicator = 0 }
		}
		set_variable = {
			var = CAS.var_nazi_modifier_negatives
			value = -0.05
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_rebellion_indicator = 1 }
		}
		set_variable = {
			var = CAS.var_nazi_modifier_negatives
			value = -0.1
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_rebellion_indicator = 2 }
		}
		set_variable = {
			var = CAS.var_nazi_modifier_negatives
			value = -0.15
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_rebellion_indicator = 3 }
		}
		set_variable = {
			var = CAS.var_nazi_modifier_negatives
			value = -0.2
		}
	}
	if = {
		limit = {
			check_variable = { var_nazi_rebellion_indicator > 3 }
		}
		set_variable = {
			var = CAS.var_nazi_modifier_negatives
			value = -0.25
		}
	}
	if = {
		limit = {
			AND = {
				check_variable = { CAS.var_nazi_rebellion_indicator_prev < CAS.var_nazi_rebellion_indicator }
				check_variable = { 5 > CAS.var_nazi_rebellion_indicator }
			}
		}
		hidden_effect = { country_event = cascadia_rework.6 }
	}
	if = {
		limit = {
			check_variable = { CAS.var_nazi_rebellion_indicator_prev > CAS.var_nazi_rebellion_indicator }
		}
		hidden_effect = { country_event = cascadia_rework.7 }
	}
}
