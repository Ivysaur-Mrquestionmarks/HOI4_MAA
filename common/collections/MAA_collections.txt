american_states_collection = {
	input = game:all_states
	name = COLLECTION_ALL_AMERICAN_STATES
	operators = {
		limit = {
			has_state_flag = part_of_america
		}
	}
}

north_america_nato_states_collection = {
	input = game:all_states
	name = COLLECTION_ALL_NA_NATO_STATES
	operators = {
		limit = {
			OR = {
				has_state_flag = part_of_america
				has_state_flag = part_of_canada
			}
		}
	}
}

faction_american_states_collection = {
	input = collection:american_states_collection

	operators = {
		limit = {
			is_controlled_by_ROOT_or_ally = yes 
		}
