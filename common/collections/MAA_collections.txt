american_states_collection = {
	input = game:all_states
	name = COLLECTION_ALL_AMERICAN_STATES
	operators = {
		limit = {
			has_state_flag = part_of_america
		}
	}
}

faction_american_states_collection = {
	input = collection:american_states_collection

	operators = {
		limit = {
			is_controlled_by_ROOT_or_ally = yes 
		}
	}
	name = COLLECTION_AMERICAN_STATES_CONTROLLED_BY_FACTION_MEMBER
}

anticommunism_american_states_collection = {
	input = collection:american_states_collection

	operators = {
		limit = {
			controller = {
				OR = {
					has_government = autocratic
					has_government = libertarian_right
					has_government = republican
				}
			}
		}
	}
}
