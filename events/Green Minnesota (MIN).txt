	desc = bomali.1.d

	fire_only_once = yes

		effect_tooltip = {
			create_corps_commander = {
				name = "Eugene Andreotti"
				skill = 3
			}
		}
		add_popularity = {
			ideology = democrat
			popularity = 0.05
		}
		add_power_balance_value = {
		    id = MIN_ventura_vs_green_army_bop
		    value = 0.075
		    tooltip_side = ventura_favored
		}
		add_war_support = -0.05
	}
	#decline
	option = {
		name = bomali.1.b
		add_popularity = {
			ideology = authoritarian_left
			popularity = 0.05
		}
		add_stability = -0.05
		hidden_effect = {
			MIL_eugene_andreotti = {
				set_nationality = MIL
			}
		}
	}
}
