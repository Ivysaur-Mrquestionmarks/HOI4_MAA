#Venezuelan Civil War begins
country_event = {
	id = poor_people.4
	title = poor_people.4.t
	desc = poor_people.4.d
	#picture =
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		450 = { set_demilitarized_zone = no }
		461 = { set_demilitarized_zone = no }
		455 = { set_demilitarized_zone = no }

		start_civil_war = {  #FOR NOW JUST HAVING IT LIKE THIS AS I DOUBT ANYONE WILL WANT TO PLAY AS VENEZUELA FOR NOW, REWORK THIS L8R
		    ideology = democrat
		    size = 0.4
		    states = {
			450 461 455
		    }
		    keep_unit_leaders_trigger = { always = yes }
		}

		set_global_flag = venezuelan_civil_war
	}
	option = {
		name = poor_people.4.a
		if = {
			limit = {
				PRA = {
					has_government = authoritarian_left
				}
			}
			add_equipment_to_stockpile = {
			    type = infantry_equipment
			    amount = 1000
			    producer = PRA
			}
		}
		every_country = {
			limit = {
				original_tag = VEN
				has_government = democrat
			}
			set_country_flag = venezuelan_liberal_rebels
			VEN_edmundo_gonzalez = {
				add_field_marshal_role = {
				  skill = 1
				  attack_skill = 1
				  defense_skill = 1
				  planning_skill = 2
				  logistics_skill = 3
				}
				add_unit_leader_trait = politically_connected
				add_unit_leader_trait = trickster
				add_unit_leader_trait = guerilla_fighter
			}
		}
		VEN_richard_vargas = {
			set_nationality = PREV
		}
		VEN_maria_corina_machado_civil_war = {
			set_nationality = PREV
		}
		hidden_effect = {
			every_country = {
				limit = {
					original_tag = VEN
					has_government = democrat
				}
				add_country_leader_role = {
				    character = VEN_maria_corina_machado_civil_war
				    promote_leader = yes
				    country_leader = {
				        ideology = trumpist
				        traits = { crowned_adder }
				    }
				}
				country_event = { id = poor_people.6 days = 35 }
			}
		}
	}
}

#Maduro wins Venezuelan Civil War - Portrait Change & Leader Traits
country_event = {
	id = poor_people.5
	title = poor_people.5.t
	desc = poor_people.5.d
	fire_only_once = yes
	trigger = {
		has_global_flag = venezuelan_civil_war
		original_tag = VEN
		has_government = authoritarian_left
		has_war = no
		has_country_leader = {
			character = VEN_nicolas_maduro
			ruling_only = yes
		}
	}
	option = {
		name = poor_people.5.a
		set_portraits = {
		    character = VEN_nicolas_maduro
		    army = { large ="GFX_portrait_maduro_postwar" }
		    civilian = { large ="GFX_portrait_maduro_postwar" }
		}
		VEN_nicolas_maduro = {
			swap_country_leader_traits = {
			    remove = dictator
			    add = VEN_super_bigote
			    ideology = communism
			}
			add_country_leader_trait = {
			    ideology = communism
			    trait = VEN_vercingetorix
			}
		}
	}
}
